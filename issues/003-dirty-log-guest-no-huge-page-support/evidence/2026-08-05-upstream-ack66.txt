{
	struct pkvm_hyp_vm *vm = pkvm_hyp_vcpu_to_hyp_vm(hyp_vcpu);
	u64 guest_addr = hyp_pfn_to_phys(gfn);
	u64 host_addr = hyp_pfn_to_phys(pfn);
	struct pkvm_mem_transition share = {
	host_lock_component();
	guest_lock_component(vm);

	ret = check_unshare(&checked_tx);
	if (ret)
		goto unlock;

	ret = kvm_pgtable_stage2_map(&vm->pgt, guest_addr, PAGE_SIZE,
				     host_addr, KVM_PGTABLE_PROT_RWX,

unlock:

## 2. klinux renamed it, dropped pfn, and re-derives host_addr with a 4 KiB-only lookup
##    arch/arm64/kvm/hyp/xhypervisor/src/mem_protect/permissions.rs:135,147
pub fn __pkvm_host_dirty_log_guest(gfn: u64, vcpu: *mut pkvm_hyp_vcpu) -> i32 {
    let mut hyp_vcpu_ref = ptr_wrapper!(vcpu);

    let mut ret =  ptr_wrapper!(vm).guest_get_valid_pte(&mut host_addr, guest_addr, 0, &mut _pte);
    if ret != 0 {
        guest_unlock_component(vm);
        host_unlock_component();
        return ret;
    }
