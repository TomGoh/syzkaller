# Guest-internal Gunyah surface (arm64 only).
#
# syz_gunyah_setup_vm$arm64 donates one parcel at GPA 0 through
# GH_VM_SET_USER_MEM_REGION and points GH_VM_SET_DTB_CONFIG at it.
# GPA 0 is the unset firmware binding, so the driver sends no demand-paging
# ranges. Further parcels at the SYZOS link addresses make VM_START fail
# with RM error 9 (MAP_FAILED): the manager's sibling donate is DENIED.
# The DTB describes a doorbell-source (label 7), a doorbell (label 4),
# a message-queue sender (label 8), a message-queue receiver (label 9) and
# a message-queue pair (label 10), so VM_INIT creates those objects.
# syz_gunyah_setup_vm_lend$arm64 is the same parcel with GH_VM_ANDROID_LEND_USER_MEM,
# which makes the image a lend and marks the VM protected.
# syz_gunyah_add_vcpu$arm64 sets the boot PC to the payload in the second
# page of that parcel, adds one proxy vCPU, arms a level irqfd on doorbell
# label 7, GH_VM_STARTs, writes the eventfd and GH_VCPU_RUNs. The payload
# issues doorbell and message-queue hypercalls, then loads an unmapped IPA.
#
# AUTH_NONE's default PC is the DTB parcel (FDT magic). The pseudo-syscall
# overrides it. Do not also issue GH_VM_ANDROID_SET_FW_CONFIG: that switches
# the VM to the protected Android auth, which rejects a PC override.

meta arches["arm64"]
meta noextract

# Opaque host pointer at the start of the donated vma, same idea as kvm_syz_vm$arm64.
resource gh_syz_vm$arm64[int64]

syz_gunyah_setup_vm$arm64(fd fd_gh_vm, usermem vma[1024]) gh_syz_vm$arm64
syz_gunyah_setup_vm_lend$arm64(fd fd_gh_vm, usermem vma[1024]) gh_syz_vm$arm64

# no_squash keeps the kvm_text_arm64 pointer intact, matching syz_kvm_add_vcpu$arm64.
# The first call starts the VM and runs vCPU 0. A second call returns EBUSY:
# Gunyah applies the boot context at GH_VM_START, and this helper starts on
# the first vCPU.
syz_gunyah_add_vcpu$arm64(vm gh_syz_vm$arm64, text ptr[in, kvm_text_arm64]) fd_gh_vcpu (no_squash)

# Raw boot-context ioctl. The helper above issues the same command before
# GH_VM_START. reg is (set << 8) | index, not a bit mask: set 0 is Xn
# (the driver accepts index 0..31; the RM accepts 0..30), set 1 is PC
# (index must be 0), set 2 is SP_ELn (index 0 = SP_EL0, index 1 = SP_EL1;
# the driver also accepts index 2, which the RM rejects).
# Values are a choice, not flags: OR-ing encodings produces an illegal reg id.
# AUTH_NONE is not an argument of this ioctl. It is the driver's default
# (GUNYAH_RM_VM_AUTH_NONE = 0) until GH_VM_ANDROID_SET_FW_CONFIG is issued,
# which this helper surface does not issue.
ioctl$GH_VM_SET_BOOT_CONTEXT(fd fd_gh_vm, cmd const[GH_VM_SET_BOOT_CONTEXT], arg ptr[in, gh_vm_boot_context])
